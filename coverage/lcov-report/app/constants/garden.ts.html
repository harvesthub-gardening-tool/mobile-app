const GRASS_STEP = 60;
const GRASS_EMOJIS = ["\ud83c\udf3f", "\u2618\ufe0f", "\ud83c\udf40"];
const GRASS_AREA = MAP_SIZE - 80; // inner area excluding border (40px each side)
 
function generateGrassDecorations() {
    const decorations: Array<{
        x: number;
        y: number;
        emoji: string;
        size: number;
    }> = [];
    let idx = 0;
    for (let y = 0; y < GRASS_AREA; y += GRASS_STEP) {
        for (let x = 0; x < GRASS_AREA; x += GRASS_STEP) {
            const seed = idx * 7919;
            const offsetX = (seed % 15) - 7;
            const offsetY = ((seed * 3) % 15) - 7;
            decorations.push({
                x: x + offsetX,
                y: y + offsetY,
                emoji: GRASS_EMOJIS[idx % GRASS_EMOJIS.length],
                size: 22 + (seed % 8),
            });
            idx++;
        }
    }
    return decorations;
}
 
export const GRASS_DECORATIONS = generateGrassDecorations();
